cmake_minimum_required(VERSION 3.22)
project(soundoff LANGUAGES C CXX)
include(FetchContent)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
add_compile_options(-O3)
FetchContent_Declare(whisper URL https://github.com/ggml-org/whisper.cpp/archive/a8d002cfd879315632a579e73f0148d06959de36.tar.gz)
FetchContent_MakeAvailable(whisper)
add_library(soundoff SHARED whisper_jni.cpp)
target_link_libraries(soundoff whisper android log)
target_compile_features(soundoff PRIVATE cxx_std_17)
target_link_options(soundoff PRIVATE "-Wl,-z,max-page-size=16384")

include("${CMAKE_CURRENT_SOURCE_DIR}/../assets/lame-build.cmake")
add_library(soundoff_mp3 SHARED mp3_jni.cpp)
target_link_libraries(soundoff_mp3 mp3lame)
target_link_options(soundoff_mp3 PRIVATE "-Wl,-z,max-page-size=16384")
